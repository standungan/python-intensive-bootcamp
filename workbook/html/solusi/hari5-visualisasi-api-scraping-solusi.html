from contextlib import contextmanager
from unittest.mock import patch

STRICT = True
_hasil = {}

@contextmanager
def periksa(kode):
    """Dipakai di sel 'Periksa jawaban': gagal -> pesan petunjuk; lolos -> tanda centang."""
    try:
        yield
        _hasil[kode] = True
        print(f"✅ {kode} benar!")
    except AssertionError as e:
        _hasil[kode] = False
        print(f"❌ {kode} belum tepat: {e}")
        if STRICT: raise
    except Exception as e:
        _hasil[kode] = False
        print(f"❌ {kode} belum bisa dijalankan: {type(e).__name__}: {e}")
        if STRICT: raise

@contextmanager
def masukan(*jawaban):
    """Simulasi input(): with masukan('Ani', '80'): ... -> input() mengembalikan 'Ani' lalu '80'."""
    with patch("builtins.input", side_effect=list(jawaban)):
        yield

def ringkasan(semua):
    """Cetak rekap latihan: ✅ benar, ❌ salah, ⏳ belum dicoba."""
    benar = 0
    for kode in semua:
        s = _hasil.get(kode)
        tanda = "⏳ belum dicoba" if s is None else ("✅ benar" if s else "❌ perlu diperbaiki")
        benar += bool(s)
        print(f"{kode:<6} {tanda}")
    print(f"\nSkor: {benar} dari {len(semua)} latihan benar")

def plot_garis(x, seri, judul):
    fig, ax = plt.subplots(figsize=(7, 4))
    for nama, y in seri.items():
        ax.plot(x, y, marker="o", label=nama)
    ax.set_title(judul)
    ax.set_xlabel("Bulan")
    ax.set_ylabel("Nilai")
    ax.legend()
    fig.tight_layout()
    return fig

def plot_bar(kategori, nilai, judul):
    fig, ax = plt.subplots(figsize=(7, 4))
    batang = ax.bar(kategori, nilai, color="steelblue", edgecolor="black")
    for b, v in zip(batang, nilai):
        ax.text(b.get_x() + b.get_width() / 2, b.get_height() + 1, str(v), ha="center")
    ax.set_title(judul)
    ax.set_xlabel("Program studi")
    ax.set_ylabel("Jumlah mahasiswa")
    fig.tight_layout()
    return fig

def plot_hist(data, bins, judul):
    fig, ax = plt.subplots(figsize=(7, 4))
    ax.hist(data, bins=bins, color="steelblue", edgecolor="white")
    rata = float(np.mean(data))
    ax.axvline(rata, color="red", linestyle="--", label=f"Rata-rata = {rata:.1f}")
    ax.set_title(judul)
    ax.set_xlabel("Nilai")
    ax.set_ylabel("Frekuensi")
    ax.legend()
    fig.tight_layout()
    return fig

import requests

def ambil_json(url, params=None, timeout=10):
    try:
        resp = requests.get(url, params=params, timeout=timeout)
        resp.raise_for_status()
        return resp.json()
    except (requests.exceptions.RequestException, ValueError) as e:
        print("Gagal:", type(e).__name__)
        return None

    from http.server import BaseHTTPRequestHandler, HTTPServer

    class _H(BaseHTTPRequestHandler):
        def log_message(self, *a): pass
        def do_GET(self):
            if self.path.startswith("/ok"):
                body, code, ctype = json.dumps({"halo": "dunia"}), 200, "application/json"
            elif self.path == "/html":
                body, code, ctype = "<html>bukan json</html>", 200, "text/html"
            elif self.path == "/lambat":
                time.sleep(1.0); body, code, ctype = "{}", 200, "application/json"
            else:
                body, code, ctype = "tidak ada", 404, "text/plain"
            self.send_response(code); self.send_header("Content-Type", ctype); self.end_headers()
            try: self.wfile.write(body.encode())
            except BrokenPipeError: pass

    _srv = HTTPServer(("127.0.0.1", 0), _H)
    threading.Thread(target=_srv.serve_forever, daemon=True).start()
    base = f"http://127.0.0.1:{_srv.server_address[1]}"
    try:

    hasil = []
    for q in soup.find_all("div", class_="quote"):
        teks = q.find("span", class_="text")
        penulis = q.find("small", class_="author")
        if teks and penulis:
            hasil.append({"penulis": penulis.get_text(), "teks": teks.get_text()})
    return hasil

import feedparser
from collections import Counter
import re

xml = '''<?xml version="1.0"?>
<feed xmlns="http://www.w3.org/2005/Atom">
 <entry><title>Deep Learning for Education</title><published>2024-03-01T00:00:00Z</published>
   <author><name>A. Contoh</name></author><author><name>B. Sampel</name></author></entry>
 <entry><title>Learning Analytics with Deep Models</title><published>2025-06-10T00:00:00Z</published>
   <author><name>C. Fiktif</name></author></entry>
 <entry><title>Deep Learning in Practice</title><published>2025-09-02T00:00:00Z</published>
   <author><name>A. Contoh</name></author></entry>
</feed>'''

def baca_feed(xml):
    feed = feedparser.parse(xml)
    baris = []
    for e in feed.entries:
        baris.append({
            "judul": " ".join(e.title.split()),
            "penulis": ", ".join(a.name for a in e.authors),
            "tahun": int(e.published[:4]),
        })
    return pd.DataFrame(baris)

def kata_sering(df, n):
    stop = {"the", "for", "and", "of", "in", "with"}
    hitung = Counter()
    for judul in df["judul"]:
        for w in re.findall(r"[a-zA-Z]{3,}", judul.lower()):
            if w not in stop:
                hitung[w] += 1
    return hitung.most_common(n)

def analisis_arxiv(xml):
    df = baca_feed(xml)
    per_tahun = df.groupby("tahun").size().sort_index()
    semua = Counter(a for s in df["penulis"] for a in s.split(", "))
    return {"per_tahun": per_tahun, "kata": kata_sering(df, 3), "penulis_teratas": semua.most_common(1)[0][0]}

def figur_eda(df):
    fig, axes = plt.subplots(2, 2, figsize=(11, 8))
    axes[0, 0].hist(df["total_bill"], bins=15, color="steelblue", edgecolor="white")
    axes[0, 0].set_title("Distribusi Total Bill")
    rata = df.groupby("day")["tip"].mean().reindex(["Thur", "Fri", "Sat", "Sun"])
    axes[0, 1].bar(rata.index, rata.values, color="coral")
    axes[0, 1].set_title("Rata-rata Tip per Hari")
    sns.scatterplot(data=df, x="total_bill", y="tip", hue="sex", ax=axes[1, 0])
    axes[1, 0].set_title("Tip vs Total Bill")
    sns.heatmap(df.select_dtypes(include="number").corr(), annot=True, fmt=".2f", cmap="coolwarm", center=0, ax=axes[1, 1])
    axes[1, 1].set_title("Korelasi")
    fig.suptitle("EDA Tips (data contoh)")
    fig.tight_layout()
    return fig

def insight(df):
